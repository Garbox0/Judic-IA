function processKeywords(keywordIds, maxRetries = 5, retryInterval = 500) {
return new Promise((resolve) => {
let attempts = 0;

function tryProcess() {
if (keywordsLoaded && storedKeywords.length > 0) {
// Keywords disponibles, procesar
const selectedKeywords = storedKeywords.filter(k => keywordIds.includes(k.id.toString()));
const keywordNames = selectedKeywords.map(k => k.name);
resolve(keywordNames);
} else if (attempts < maxRetries) {

setTimeout(tryProcess, retryInterval);

resolve([]);
}
}

tryProcess();
});

// Función para calcular conteos actualizados basándose en la jerarquía
function calculateUpdatedCounts(fieldType, lowestSelectedFilter, state, level, jurisdiction, courtType, courtFull) {
console.log('calculateUpdatedCounts called with:', {
fieldType, lowestSelectedFilter, state, level, jurisdiction, courtType, courtFull
});
let originalData = [];
let fieldName = '';
// Obtener los datos originales según el tipo de campo
switch(fieldType) {
case 'state':
originalData = currentResponse.states;
fieldName = 'name';
break;
case 'level':
originalData = currentResponse.levels;

break;

break;
case 'court_type':

fieldName = 'name';

}

return originalData.map(item => {

updatedCount = calculatePropagatedCount(fieldType, item[fieldName], state, level, jurisdiction, courtType, courtFull);

};
}).filter(item => item.count > 0); // Solo mostrar opciones con conteo > 0
}

// Función para calcular conteos propagados desde el nivel más bajo
function calculatePropagatedCount(fieldType, fieldValue, state, level, jurisdiction, courtType, courtFull) {
let totalCount = 0;
// Si no hay filtros aplicados, usar el conteo original
if (!state && !level && !jurisdiction && !courtType && !courtFull) {
return getOriginalCount(fieldType, fieldValue);
}
// Si este campo está seleccionado, usar su conteo original
if ((fieldType === 'state' && state === fieldValue) ||
(fieldType === 'level' && level === fieldValue) ||
(fieldType === 'jurisdiction' && jurisdiction === fieldValue) ||
(fieldType === 'court_type' && courtType === fieldValue) ||
(fieldType === 'court_full' && courtFull === fieldValue)) {
return getOriginalCount(fieldType, fieldValue);
}
// Recorrer toda la jerarquía
Object.keys(currentHierarchy).forEach(stateKey => {
// Si hay filtro de provincia y no coincide, saltar
if (state && stateKey !== state) return;
const stateData = currentHierarchy[stateKey];
if (!stateData) return;
Object.keys(stateData).forEach(levelKey => {
// Si hay filtro de instancia y no coincide, saltar
if (level && levelKey !== level) return;
const levelData = stateData[levelKey];
if (!levelData) return;
Object.keys(levelData).forEach(jurisdictionKey => {
// Si hay filtro de jurisdicción y no coincide, saltar
if (jurisdiction && jurisdictionKey !== jurisdiction) return;
const jurisdictionData = levelData[jurisdictionKey];
if (!jurisdictionData) return;
Object.keys(jurisdictionData).forEach(courtTypeKey => {
// Si hay filtro de fuero y no coincide, saltar
if (courtType && courtTypeKey !== courtType) return;
const courtTypeData = jurisdictionData[courtTypeKey];
if (!courtTypeData) return;
// Para cada tribunal en este fuero
courtTypeData.forEach(courtFullName => {
// Si hay filtro de tribunal y no coincide, saltar
if (courtFull && courtFullName !== courtFull) return;
// Verificar si este tribunal contribuye al campo que estamos calculando
let shouldCount = false;
switch(fieldType) {
case 'state':
shouldCount = stateKey === fieldValue;
break;
case 'level':
shouldCount = levelKey === fieldValue;
break;
case 'jurisdiction':
shouldCount = jurisdictionKey === fieldValue;
break;
case 'court_type':
shouldCount = courtTypeKey === fieldValue;
break;
case 'court_full':
shouldCount = courtFullName === fieldValue;
break;
}
if (shouldCount) {
// Sumar el conteo original de este tribunal
totalCount += getOriginalCount('court_full', courtFullName);
}
});
});
});
});
});
// Si no encontramos ningún tribunal que contribuya, verificar si el campo existe en la jerarquía
if (totalCount === 0) {
// Verificar si el campo existe en la jerarquía para los filtros aplicados
if (fieldExistsInHierarchy(fieldType, fieldValue, state, level, jurisdiction, courtType, courtFull)) {
// Si existe pero no tiene tribunales, usar el conteo original
return getOriginalCount(fieldType, fieldValue);
}
// Si no existe en la jerarquía, intentar calcular basándose en los padres
let parentCount = calculateCountFromParents(fieldType, fieldValue, state, level, jurisdiction, courtType, courtFull);
if (parentCount > 0) {
return parentCount;
}
}
return totalCount;
}

// Función para verificar si un campo existe en la jerarquía con los filtros aplicados
function fieldExistsInHierarchy(fieldType, fieldValue, state, level, jurisdiction, courtType, courtFull) {
return Object.keys(currentHierarchy).some(stateKey => {
if (state && stateKey !== state) return false;
const stateData = currentHierarchy[stateKey];
if (!stateData) return false;
return Object.keys(stateData).some(levelKey => {
if (level && levelKey !== level) return false;
const levelData = stateData[levelKey];
if (!levelData) return false;
return Object.keys(levelData).some(jurisdictionKey => {
if (jurisdiction && jurisdictionKey !== jurisdiction) return false;
const jurisdictionData = levelData[jurisdictionKey];
if (!jurisdictionData) return false;
return Object.keys(jurisdictionData).some(courtTypeKey => {
if (courtType && courtTypeKey !== courtType) return false;
const courtTypeData = jurisdictionData[courtTypeKey];
if (!courtTypeData) return false;
// Verificar si el campo coincide
switch(fieldType) {
case 'state':
return stateKey === fieldValue;
case 'level':
return levelKey === fieldValue;
case 'jurisdiction':
return jurisdictionKey === fieldValue;
case 'court_type':
return courtTypeKey === fieldValue;
case 'court_full':
return courtTypeData.includes(fieldValue);
default:
return false;
}
});
});
});
});
}
